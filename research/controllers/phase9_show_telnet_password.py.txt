"""Owner-requested local display only. Never print or persist the credential."""
import ctypes
import json
import os
from pathlib import Path

base = Path(os.environ['LOCALAPPDATA']) / 'DLinkResearch'
try:
    manifest = json.loads((base / 'firmware/filesystem-manifest.json').read_text())
    binary = Path(manifest['/usr/sbin/ams_cli']['file']).read_bytes()
    account = binary[0x18864:0x18a64].split(b'\0', 1)[0]
    secret = binary[0x18a64:0x18c64].split(b'\0', 1)[0]
    if account != b'root' or not 0 < len(secret) < 64:
        raise ValueError('Unexpected credential artifact')
    ctypes.windll.user32.MessageBoxW(
        None,
        'Previously verified Telnet management account: root\n\n'
        + secret.decode('ascii')
        + '\n\nType this into the local setup password prompt.\nDo not paste it into chat. Close this dialog afterward.',
        'D-Link Telnet credential — local display requested by owner',
        0x00040000,
    )
except Exception:
    ctypes.windll.user32.MessageBoxW(None, 'Unable to validate the existing local credential artifact. No credential displayed.', 'D-Link local credential lookup', 0x10)
