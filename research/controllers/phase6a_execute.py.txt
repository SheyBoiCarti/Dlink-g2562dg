def save():

def connect():
    console = Console()
    assert 'LOGIN#' in console.read()
    console.sock.sendall(namespace['account'] + b'\r\n')
    assert 'PASS#' in console.read()
    console.sock.sendall(namespace['secret'] + b'\r\n')
    assert 'AMS#' in console.read()
    console.send('sh')
    assert '#' in console.read()
    console.send('unset HISTFILE; HISTSIZE=0')
    console.read()
    return console


def run(label, command, timeout=20, record=True, check=True):
    global counter
    assert '\n' not in command and '\r' not in command
    counter += 1
    begin = '__P6_BEGIN_%d__' % counter
    code = '__P6_CODE_%d__' % counter
    end = '__P6_END_%d__' % counter
    c.send("printf '\\n" + begin + "\\n'; " + command +
           "; p6_rc=$?; printf '\\n" + code + "%s\\n" + end + "\\n' \"$p6_rc\"")
    raw = c.read(timeout, until=('\n' + end + '\n').encode())
    match = re.search(r'(?ms)^' + begin + r'\n(.*?)\n' + code + r'(\d+)\n' + end, raw)
    if not match:
        raise RuntimeError('Missing completion frame: ' + label)
    output = sanitize(match[1]).strip()
    rc = int(match[2])
    if record:
        result['records'].append({'name': label, 'exit': rc, 'output': output})
        save()
        print(label + ':\n' + output, flush=True)
    if check and rc:
        raise RuntimeError(label + ' returned ' + str(rc))
    return output, rc


def upload(local, remote):
    data = local.read_bytes().replace(b'\r\n', b'\n')
    encoded = base64.b64encode(data).decode()
    run('initialize upload', ': > ' + shlex.quote(remote + '.b64'), record=False)
    for i in range(0, len(encoded), 180):
        run('upload block', "printf '%s' " + shlex.quote(encoded[i:i+180]) +
            ' >> ' + shlex.quote(remote + '.b64'), record=False)
    command = ('base64 -d ' + shlex.quote(remote + '.b64') + ' > ' + shlex.quote(remote) +
               ' && chmod 700 ' + shlex.quote(remote) + ' && rm ' + shlex.quote(remote + '.b64') +
               ' && md5sum ' + shlex.quote(remote))
    output, _ = run('verify uploaded control file', command, record=False)
    assert hashlib.md5(data).hexdigest() in output
    result['records'].append({'name': 'control file staged', 'file': remote,
                              'verified_against_local_file': True})
    save()


def ssh_check(pty):
    args = [r'C:\Program Files\Git\usr\bin\ssh.exe', '-v', '-F', 'none',
            '-p', '2222', '-i', str(KEYDIR / 'id_rsa_dlink_phase6a'),
            '-o', 'UserKnownHostsFile=' + str(KEYDIR / 'known_hosts_phase6a'),
            '-o', 'GlobalKnownHostsFile=none', '-o', 'StrictHostKeyChecking=yes',
            '-o', 'UpdateHostKeys=no', '-o', 'IdentitiesOnly=yes',
            '-o', 'IdentityAgent=none', '-o', 'PreferredAuthentications=publickey',
            '-o', 'PasswordAuthentication=no', '-o', 'KbdInteractiveAuthentication=no',
            '-o', 'BatchMode=yes', '-o', 'ConnectTimeout=5',
            '-o', 'ConnectionAttempts=1', '-o', 'KexAlgorithms=+diffie-hellman-group14-sha1',
            '-o', 'HostKeyAlgorithms=+ssh-rsa', '-o', 'PubkeyAcceptedAlgorithms=+ssh-rsa',
            '-tt' if pty else '-T', 'admin@10.0.0.2',
            "unset HISTFILE; printf 'PHASE6A_SSH_PROOF\\n'; id -u; id; uname -srm"]
    completed = subprocess.run(args, capture_output=True, text=True, timeout=25)
    output = sanitize(completed.stdout.replace('\r', ''))
    retained = [line for line in completed.stderr.splitlines()
                if any(word in line for word in ('Authenticated to ', 'Authentication succeeded',
                    'Server accepts key:', 'signing using ', 'Permission denied',
                    'no mutual signature', 'known and matches', 'PTY allocation request failed',
                    'Host key verification failed', 'Connection refused', 'Load key ',
                    'channel 0:', 'Exit status ', 'pty', 'request failed', 'Connection to ', 'error', 'failed'))]
    proof = {'pty_requested': pty, 'exit': completed.returncode,
             'stdout': output, 'ssh_evidence': retained,
             'uid0_verified': bool(re.search(r'uid=0\b', output)),
             'authentication_publickey_verified': any('Authenticated to ' in line and
                                                      'publickey' in line for line in retained)}
    result['ssh_checks'].append(proof)
    save()
    print('SSH CHECK ' + ('PTY' if pty else 'EXEC') + ':\n' + output +
          '\n' + '\n'.join(retained), flush=True)
    assert proof['exit'] == 0 and proof['uid0_verified']
    assert proof['authentication_publickey_verified']
    assert 'Linux 3.14.18 armv7l' in output
